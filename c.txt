import java.util.*;
public class Hill {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter 2 letters: ");
        String text = sc.nextLine().toUpperCase();
        int a=3,b=3,c=2,d=5;
        int x = text.charAt(0)-65;
        int y = text.charAt(1)-65;
        // Encryption
        int e1 = (a*x + b*y) % 26;
        int e2 = (c*x + d*y) % 26;
        System.out.println("Encrypted: " + (char)(e1+65) + "" + (char)(e2+65));
        // Decryption (using inverse matrix)
        int p1 = (15*e1 + 17*e2) % 26;
        int p2 = (20*e1 + 9*e2) % 26;
        System.out.println("Decrypted: " + (char)(p1+65) + "" + (char)(p2+65));
    }
}
