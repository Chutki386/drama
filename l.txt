Playfair cipher
import java.util.*;
class Playfair {
                           static char[][] m=new char[5][5];
static void gen(String k){
String key="";
for(char c:(k+"ABCDEFGHIKLMNOPQRSTUVWXYZ").toCharArray())
if(c!='J' && key.indexOf(c)==-1) key+=c;
for(int i=0;i<25;i++) m[i/5][i%5]=key.charAt(i);
}
static String process(String t,int dir){
String r="";
for(int i=0;i<t.length();i+=2){
char a=t.charAt(i), b=t.charAt(i+1);
int r1=0,c1=0,r2=0,c2=0;
for(int i1=0;i1<5;i1++)
for(int j=0;j<5;j++){
if(m[i1][j]==a){r1=i1;c1=j;}
if(m[i1][j]==b){r2=i1;c2=j;}
}
if(r1==r2) r+=m[r1][(c1+dir+5)%5]+""+m[r2][(c2+dir+5)%5];
else if(c1==c2) r+=m[(r1+dir+5)%5][c1]+""+m[(r2+dir+5)%5][c2];
else r+=m[r1][c2]+""+m[r2][c1];
}
return r;
}
public static void main(String[] a){
Scanner s=new Scanner(System.in);
System.out.print("Key: ");
gen(s.nextLine().toUpperCase().replaceAll("J","I"));       System.out.print("Text: ");
String t=s.nextLine().toUpperCase().replaceAll("J","I").replaceAll(" ","");
if(t.length()%2!=0) t+="X";
String e=process(t,1);
System.out.println("Encrypted: "+e);
System.out.println("Decrypted: "+process(e,-1));
}
}
